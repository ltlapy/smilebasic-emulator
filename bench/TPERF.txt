'TPERF: A REPEATABLE PERFORMANCE BENCHMARK FOR SMILEBASIC 4
'RUN IT, WAIT ABOUT 4 MINUTES, ANSWER THE SAVE QUESTION. NO OTHER INPUT.
'EACH SCENE ISOLATES ONE KIND OF WORK. WORK IS THE TIME THE PROGRAM SPENDS
'BETWEEN TWO VSYNCS, FRAME THE TIME FROM VSYNC TO VSYNC, DRAWING INCLUDED.
OPTION STRICT
'THE SCENE TO RUN ALONE (SEE THE LIST AT @SCENES, FROM 1), 0 FOR ALL
CONST #ONLY=0
CONST #FRAMES=300
'SCALES EVERY COUNT BELOW; COMPARE ONLY RUNS WITH THE SAME VALUE
CONST #LOADPCT=100
'A NOTE SAVED WITH THE RESULTS, SUCH AS "SWITCH1"
CONST #TAG=""
CONST #RESFILE="RESPERF"
'A SCENE ENDS EARLY AFTER THIS LONG, SO THAT A SLOW MACHINE STILL FINISHES
CONST #SCENEMS=20000
'
'WHAT EACH SCENE DOES IN A FRAME. SIZED FROM A SWITCH'S 35000 TURNS OF A
'SIMPLE TWO-STATEMENT LOOP A FRAME; TUNE AFTER SEEING A SWITCH'S RESULTS.
CONST #NVINT=18000,#NVREAL=9000,#NVSTR=3000,#NVARR=10000
CONST #NVDEF=4000,#NVVAR=5000
CONST #NSPMOVE=2500,#NSPROT=1200,#NSPCHURN=800,#NSPCHPOOL=3000
CONST #NSPHIT=1000,#NSPHITQ=300,#NSPANIM=3000,#NSPANIMR=100
CONST #NTPRINT=3,#NTPUT=6000,#NTSCROLL=100,#NTFILL=2000
CONST #NGFILL=2000,#NGALPHA=1200,#NGLINE=3000,#NGCIRCLE=1500,#NGTRI=1500
CONST #NGCHR=600,#NGCOPY=900,#NGLOAD=6,#NGUPLOAD=3,#NGARRAY=10000
CONST #NLCOMP=7,#NLBLUR=3,#NLMOSAIC=3,#NLCOLOR=4,#NLMATRIX=4,#NLRASTER=2
'THE RING OF THE GAME SCENE: VERTICES ARE U*V, TRIANGLES TWICE THAT
CONST #GMU=30,#GMV=20,#NGMSP=200,#NGMTRI=1200,#NGMSPAWN=600
'THE MAX LOAD SCENES: FIRST COUNT, UPPER LIMIT, FRAMES A STEP, MS A STEP,
'MS A SCENE, AND THE MEAN FRAME IN MICROSECONDS THAT STILL COUNTS AS 60 FPS
CONST #MAXSP0=200,#MAXSPCAP=4000,#MAXTRI0=200,#MAXTRICAP=100000
CONST #RAMPFRAMES=30,#RAMPSTEPMS=2000,#RAMPMS=30000,#RAMPOK=17500
'THE GAUGE'S SPRITES; SCENES STAY BELOW THESE
CONST #GWORK=4094,#GFRAME=4093,#GTICK=4092
CONST #SCENES=34,#NR=8192
'
VAR NM$[#SCENES],WORK%[#FRAMES],FRM%[#FRAMES]
VAR RN,RNAME$[#SCENES],RCNT[#SCENES],RFR[#SCENES],RLOST[#SCENES],RWALL[#SCENES]
VAR RWM#[#SCENES],RWP#[#SCENES],RWX#[#SCENES]
VAR RFM#[#SCENES],RFP#[#SCENES],RFX#[#SCENES]
VAR FC,CUR,T0,M0,FLIM,MSLIM,LASTF,COUNT
VAR RX%[#NR],RY%[#NR],RC%[#NR],SINT#[256]
VAR FROZEN,TOTALMS
VAR MLO,MHI,MCUR,MCAP,MT0
VAR GVA,GVB,GVC,GVD
VAR PAGE,PAGES,SAVED$
VAR MI
'
RESTORE @SCENES
FOR MI=0 TO #SCENES-1
 READ NM$[MI]
NEXT
BOOT
FOR MI=0 TO #SCENES-1
 IF #ONLY==0 || #ONLY==MI+1 THEN
  CUR=MI
  CALL "SC"+NM$[MI]
 ENDIF
NEXT
TOTALMS=MILLISEC()-TOTALMS
FINISHRUN
END
'
@SCENES
DATA "VINT","VREAL","VSTR","VARRAY","VDEF","VVAR"
DATA "SPMOVE","SPROT","SPCHURN","SPHIT","SPANIM"
DATA "TPRINT","TPUT","TSCROLL","TMULTI"
DATA "GFILL","GALPHA","GLINE","GCIRCLE","GTRI"
DATA "GPUTCHR","GCOPY","GLOAD","GUPLOAD","GARRAY"
DATA "LCOMP","LBLUR","LMOSAIC","LCOLOR","LMATRIX","LRASTER"
DATA "GAME","MAXSP","MAXTRI"
'
'SET-UP THAT ONLY HAPPENS ONCE
DEF BOOT
 VAR I
 ACLS
 ?"TPERF: PREPARING. NO INPUT NEEDED."
 RANDOMIZE 1,&H51EB
 FOR I=0 TO #NR-1
  RX%[I]=RND(1,400)
  RY%[I]=RND(1,240)
  RC%[I]=RGB(RND(1,256),RND(1,256),RND(1,256))
 NEXT
 FOR I=0 TO 255
  SINT#[I]=SIN(I*#PI/128)
 NEXT
 MAKETEX
 'A TIMER THAT GIVES 0 HERE IS A RUN WITH FROZEN TIME (--FIXED): MAX LOAD
 'HAS NOTHING TO GO BY THERE
 PERFBEGIN 7
 FOR I=1 TO 3000
 NEXT
 FROZEN=PERFEND(7)<=0
 GVA=1:GVB=2:GVC=3:GVD=4
 TOTALMS=MILLISEC()
END
'
'GRP3 HOLDS THE PROGRAM'S OWN SPRITE IMAGES, AND STAYS THROUGH THE SCENES
DEF MAKETEX
 VAR I,J,X,Y
 GTARGET 3
 GFILL 0,0,7,7,#C_WHITE
 'EIGHT 16X16 IMAGES IN ROW 16
 FOR I=0 TO 7
  X=I*16:Y=16
  GFILL X+1,Y+1,X+14,Y+14,RGB(255,80+I*24,255-I*24,128)
  GBOX X+1,Y+1,X+14,Y+14,#C_WHITE
  GCIRCLE X+8,Y+8,4,#C_BLACK
 NEXT
 'FOUR 32X32 IMAGES IN ROW 32
 FOR I=0 TO 3
  X=I*32:Y=32
  GCIRCLE X+16,Y+16,14,0,359,0,#C_WHITE
  GTRI X+16,Y+3,X+29,Y+28,X+3,Y+28,RGB(200,255-I*60,60+I*60,200)
 NEXT
 'A 64X64 IMAGE IN ROW 64
 FOR J=0 TO 7
  FOR I=0 TO 7
   IF ((I+J) AND 1)==0 THEN GFILL I*8,64+J*8,I*8+7,64+J*8+7,RGB(255,I*36,J*36,200)
  NEXT
 NEXT
 GTARGET 0
END
'
DEF CN(BASE)
 RETURN MAX(1,INT(BASE*#LOADPCT/100))
END
'
'RESET BEFORE A SCENE'S OWN SET-UP
DEF PREP
 VAR I
 ACLS 1,1,1,1
 FOR I=0 TO 2
  GTARGET I
  GCLS
 NEXT
 GTARGET 0
 SPPAGE 3
 RANDOMIZE 1,&H1234+CUR
END
'
'THE SCENE'S NAME AND GAUGE; THE CLOCK STARTS HERE
DEF GO CNT
 COUNT=CNT
 LOCATE 0,0
 ?FORMAT$("%2D/%2D ",CUR+1,#SCENES);NM$[CUR];" N=";CNT
 SPPAGE 3
 SPLAYER 0
 GAUGEBAR #GWORK,18,#C_LIME
 GAUGEBAR #GFRAME,23,#C_CYAN
 GAUGEBAR #GTICK,16,#C_WHITE
 SPSCALE #GTICK,0.25,3
 SPOFS #GTICK,100,16,-4000
 FLIM=#FRAMES
 MSLIM=#SCENEMS
 STARTCLOCK
END
'
DEF GAUGEBAR ID,Y,C
 SPSET ID,0,0,8,4,0
 SPLAYER ID,0
 SPPAGE ID,3
 SPOFS ID,0,Y,-4000
 SPCOLOR ID,C
 SPSCALE ID,0.01,1
END
'
'WAIT AFTER SET-UP, WHICH MAY TAKE MANY FRAMES, AND START COUNTING
DEF STARTCLOCK
 WAIT 2
 FC=-1
 LASTF=0
 T0=MILLISEC()
 M0=MAINCNT()
 PERFBEGIN 1
END
'
'CALLED IN WHILE AROUND EACH FRAME'S WORK: ENDS THE FRAME BEFORE, GIVES 0
'WHEN THE SCENE IS OVER, AND ELSE STARTS THE NEXT FRAME
DEF TF()
 VAR W,F
 IF FC>=0 THEN
  W=PERFEND(0)
  WORK%[FC]=W
  SPSCALE #GWORK,MAX(0.01,MIN(47,W*0.00075)),1
  SPSCALE #GFRAME,MAX(0.01,MIN(47,LASTF*0.00075)),1
  IF (FC MOD 30)==0 THEN
   LOCATE 0,1
   ?FORMAT$("WORK %7.2F MS  FRAME %7.2F MS  ",W/1000,LASTF/1000);
  ENDIF
  VSYNC
  F=PERFEND(1)
  PERFBEGIN 1
  FRM%[FC]=F
  LASTF=F
 ENDIF
 INC FC
 IF FC>=FLIM || MILLISEC()-T0>MSLIM THEN RETURN 0
 PERFBEGIN 0
 RETURN 1
END
'
DEF STATS A,N OUT M#,P#,X#
 VAR I,S#
 FOR I=0 TO N-1
  S#=S#+A[I]
 NEXT
 M#=S#/N/1000
 SORT 0,N,A
 P#=A[MIN(N-1,N*95 DIV 100)]/1000
 X#=A[N-1]/1000
END
'
'THE CLOCK STOPS: KEEPS THE SCENE'S ROW OF THE TABLE
DEF FIN
 VAR N=FC,R=RN,A#,B#,C#
 RLOST[R]=MAINCNT()-M0-N
 RWALL[R]=MILLISEC()-T0
 RNAME$[R]=NM$[CUR]
 RCNT[R]=COUNT
 RFR[R]=N
 STATS WORK%,N OUT A#,B#,C#
 RWM#[R]=A#:RWP#[R]=B#:RWX#[R]=C#
 STATS FRM%,N OUT A#,B#,C#
 RFM#[R]=A#:RFP#[R]=B#:RFX#[R]=C#
 INC RN
END
'
'--- VM ---
DEF SCVINT
 VAR N=CN(#NVINT),I,A%,B%
 PREP
 GO N
 WHILE TF()
  FOR I=1 TO N
   A%=(A%+I*3) AND 1023
   B%=B% XOR (A%<<1)
  NEXT
 WEND
 FIN
END
'
DEF SCVREAL
 VAR N=CN(#NVREAL),I,X#,Y#
 PREP
 GO N
 WHILE TF()
  FOR I=1 TO N
   X#=X#*0.999+I*0.25
   Y#=SQR(X#)*0.01+SIN(Y#)
  NEXT
  X#=0
 WEND
 FIN
END
'
DEF SCVSTR
 VAR N=CN(#NVSTR),I,J,S$,T$
 PREP
 GO N
 WHILE TF()
  FOR I=1 TO N
   S$=S$+CHR$(65+(I AND 15))+"XY"
   IF LEN(S$)>=120 THEN
    T$=MID$(S$,I AND 63,32)
    J=INSTR(S$,"XYQ")
    S$=STR$(I)+T$
   ENDIF
  NEXT
 WEND
 FIN
END
'
DEF SCVARRAY
 VAR N=CN(#NVARR),I,A%[4096],M#[64,64]
 PREP
 GO N
 WHILE TF()
  FOR I=1 TO N
   A%[I AND 4095]=(A%[(I*7) AND 4095]+I) AND 65535
   M#[I AND 63,(I>>6) AND 63]=M#[(I>>3) AND 63,I AND 63]*0.5+0.5
  NEXT
 WEND
 FIN
END
'
DEF ADDB(A,B)
 RETURN (A+B) AND 255
END
'
DEF TWOB A,B OUT C,D
 C=B:D=A
END
'
DEF SCVDEF
 VAR N=CN(#NVDEF),I,R,S,T
 PREP
 GO N
 WHILE TF()
  FOR I=1 TO N
   R=ADDB(I,R)
   TWOB R,I OUT S,T
  NEXT
 WEND
 FIN
END
'
DEF SCVVAR
 VAR N=CN(#NVVAR),I,R,VN$[4]
 VN$[0]="GVA":VN$[1]="GVB":VN$[2]="GVC":VN$[3]="GVD"
 PREP
 GO N
 WHILE TF()
  FOR I=1 TO N
   R=VAR(VN$[I AND 3])
   VAR(VN$[(I+1) AND 3])=R
  NEXT
 WEND
 FIN
END

'--- SPRITES ---
DEF SCSPMOVE
 VAR N=MIN(4000,CN(#NSPMOVE)),I,X#,Y#
 VAR PX#[N],PY#[N],DX#[N],DY#[N]
 PREP
 FOR I=0 TO N-1
  SPSET I,(I AND 7)*16,16,16,16,0
  PX#[I]=RX%[I]
  PY#[I]=RY%[I]
  DX#[I]=(RND(1,41)-20)/10
  DY#[I]=(RND(1,41)-20)/10
 NEXT
 GO N
 WHILE TF()
  FOR I=0 TO N-1
   X#=PX#[I]+DX#[I]
   Y#=PY#[I]+DY#[I]
   IF X#<0 || X#>384 THEN DX#[I]=-DX#[I]
   IF Y#<0 || Y#>224 THEN DY#[I]=-DY#[I]
   PX#[I]=X#:PY#[I]=Y#
   SPOFS I,X#,Y#
  NEXT
 WEND
 FIN
END
'
DEF SCSPROT
 VAR N=MIN(4000,CN(#NSPROT)),I,J,S#
 PREP
 FOR I=0 TO N-1
  SPSET I,(I AND 3)*32,32,32,32,0
  SPHOME I,16,16
  SPOFS I,RX%[I],RY%[I]
 NEXT
 GO N
 WHILE TF()
  FOR I=0 TO N-1
   J=FC+I
   S#=1+0.5*SINT#[J AND 255]
   SPROT I,(J*3) MOD 360
   SPSCALE I,S#,S#
   SPOFS I,RX%[I]+(J AND 15),RY%[I]
  NEXT
 WEND
 FIN
END
'
DEF SCSPCHURN
 VAR N=CN(#NSPCHURN),POOL=MIN(4000,CN(#NSPCHPOOL)),I,J,K,S
 PREP
 FOR I=0 TO POOL-1
  SPSET I,(I AND 7)*16,16,16,16,0
  SPOFS I,RX%[I],RY%[I]
 NEXT
 GO N
 WHILE TF()
  FOR J=0 TO N-1
   I=K MOD POOL
   INC K
   SPCLR I
   S=SPSET(0,POOL-1,(J AND 7)*16,16,16,16,0)
   SPOFS S,RX%[K AND 8191],RY%[K AND 8191]
   SPCOLOR S,RC%[K AND 8191]
  NEXT
 WEND
 FIN
END
'
DEF SCSPHIT
 VAR N=MIN(4000,CN(#NSPHIT)),Q=CN(#NSPHITQ),I,J,H,C
 PREP
 FOR I=0 TO N-1
  SPSET I,(I AND 7)*16,16,16,16,0
  SPOFS I,RX%[I],RY%[I]
  SPCOL I,0,0,16,16
 NEXT
 GO N
 WHILE TF()
  FOR J=0 TO Q-1
   I=(J*7+FC) MOD N
   SPOFS I,RX%[(I+FC) AND 8191],RY%[(I+FC) AND 8191]
   H=SPHITSP(I)
   WHILE H>=0
    INC C
    H=SPHITSP()
   WEND
  NEXT
 WEND
 FIN
END

  FOR J=0 TO 158

   FOR Y=0 TO 89
    LOCATE 0,0,Y
    COLOR 0,RC%[(Y+FC) AND 8191]
    TPRINT 0,L$[(Y+P+FC) AND 15];
   NEXT
  NEXT
 WEND
 FIN
END

'
'A DOT IN OPPOSITE CORNERS MAKES A GRAPHIC PAGE CHANGED FROM END TO END;
'THE PAGES SHOW THROUGH SPRITES, SO EACH HAS TO REACH THE SCREEN
DEF SCGUPLOAD
 VAR N=MIN(3,CN(#NGUPLOAD)),I,C

  SPSET 10+I,0,0,400,240,0
  SPPAGE 10+I,I
  SPOFS 10+I,0,0,1000+I
 NEXT

  FOR I=0 TO N-1
   GTARGET I
   GPSET FC AND 255,20,C
   GPSET 2047,2047,C
   GPSET 0,0,C
  NEXT
  GTARGET 0

'
'--- LAYERS ---
'GRP1 HOLDS A 1280X720 PICTURE THAT IS PARTLY CLEAR. EACH LAYER 1 TO K GETS
'THE PICTURE AS A SPRITE AND A FEW SMALL SPRITES THAT MOVE, SO THAT IT
'CHANGES EACH FRAME.
DEF LAYERSET K
 VAR L,I,S
 XSCREEN 1280,720
 GTARGET 1
 FOR I=0 TO 999
  GFILL RX%[I]*3,RY%[I]*3,RX%[I]*3+47,RY%[I]*3+47,RC%[I]
 NEXT
 GTARGET 0
 FOR L=1 TO K
  S=(L-1)*40
  SPSET S,0,0,1280,720,0
  SPPAGE S,1
  SPLAYER S,L
  SPOFS S,0,0,100
  FOR I=1 TO 32
   SPSET S+I,(I AND 3)*32,32,32,32,0
   SPPAGE S+I,3
   SPLAYER S+I,L
  NEXT
 NEXT
END
'
DEF LAYERMOVE K
 VAR L,I,S,J
 FOR L=1 TO K
  S=(L-1)*40
  FOR I=1 TO 32
   J=(S+I+FC*3) AND 8191
   SPOFS S+I,RX%[J]*3,RY%[J]*3
  NEXT
 NEXT
END

 GO N
 WHILE TF()
  LAYERMOVE N
  FOR L=1 TO N
   LFILTER L,1,2+((FC DIV 4+L*3) AND 31)
  NEXT
 WEND
 FIN
END
'
DEF SCLCOLOR
 VAR N=MIN(7,CN(#NLCOLOR)),L
 PREP
 LAYERSET N
 FOR L=1 TO N
  LAYER L,1
 NEXT
 GO N
 WHILE TF()
  LAYERMOVE N
  FOR L=1 TO N
   LFILTER L,5,(FC*3+L*40) MOD 360,-40,10
  NEXT
 WEND
 FIN
END
'
DEF SCLMATRIX
 VAR N=MIN(7,CN(#NLMATRIX)),L,M#[16],A#,C#,S#,Z#
 PREP
 LAYERSET N
 FOR L=1 TO N
  LAYER L,1
 NEXT
 M#[10]=1
 M#[15]=1
 Z#=0.9
 GO N
 WHILE TF()
  LAYERMOVE N
  FOR L=1 TO N
   A#=FC*0.01+L*0.3
   C#=COS(A#):S#=SIN(A#)
   M#[0]=C#*Z#/640
   M#[4]=-S#*Z#/640
   M#[12]=(S#*360/640-C#)*Z#
   M#[1]=-S#*Z#/360
   M#[5]=-C#*Z#/360
   M#[13]=(S#*640/360+C#)*Z#
   M#[3]=0.0003*SINT#[(FC+L*30) AND 255]
   LMATRIX L,M#
  NEXT
 WEND
 FIN
END
'
DEF SCLRASTER
 VAR N=MIN(7,CN(#NLRASTER)),L,I,A#[720,4],B#[320,2]
 PREP
 LAYERSET N
 FOR L=1 TO N
  LAYER L,1
 NEXT
 FOR I=0 TO 719
  A#[I,1]=1
  A#[I,3]=1
 NEXT
 FOR I=0 TO 319
  B#[I,1]=1
 NEXT
 GO N
 WHILE TF()
  LAYERMOVE N
  FOR I=0 TO 719
   A#[I,0]=SINT#[(I+FC*3) AND 255]*0.03
  NEXT
  FOR I=0 TO 319
   B#[I,0]=SINT#[(I+FC*2) AND 255]*0.03
  NEXT
  FOR L=1 TO N
   IF (L AND 1)==1 THEN
    LFILTER L,3,A#,#TRUE
   ELSE
    LFILTER L,4,B#,#TRUE
   ENDIF
  NEXT
 WEND
 FIN
END
'
'--- THE GAME SCENE ---
'A RING THAT TURNS: EVERY VERTEX TRANSFORMED IN BASIC, FACES TURNED AWAY
'LEFT OUT, THE REST SORTED AND FILLED, TWO PAGES IN TURN. SPRITES AND TEXT
'ON TOP. EVERY SO OFTEN COMES A BURST OF EXTRA WORK.
DEF SCGAME
 VAR NU=#GMU,NV=#GMV,NVT=NU*NV,NFA=NVT*2,NSP=CN(#NGMSP)
 VAR BT=CN(#NGMTRI),BS=MIN(2000,CN(#NGMSPAWN))
 VAR VX#[NVT],VY#[NVT],VZ#[NVT],SX#[NVT],SY#[NVT],SZ#[NVT]
 VAR FA%[NFA],FB%[NFA],FD%[NFA],FK%[NFA],DEPTH%[NFA],ORD%[NFA]
 VAR I,J,K,U,V,A,B,D,C,P,X,Y,E,BLUR
 VAR AU#,AV#,R#,X0#,Y0#,Z0#,X1#,Y1#,Z1#,W#,CA#,SA#,CB#,SB#
 PREP
 'THE RING
 FOR U=0 TO NU-1
  AU#=U*2*#PI/NU
  FOR V=0 TO NV-1
   AV#=V*2*#PI/NV
   R#=70+28*COS(AV#)
   I=U*NV+V
   VX#[I]=R#*COS(AU#)
   VY#[I]=28*SIN(AV#)
   VZ#[I]=R#*SIN(AU#)
   J=U*NV+((V+1) MOD NV)
   K=((U+1) MOD NU)*NV+V
   E=((U+1) MOD NU)*NV+((V+1) MOD NV)
   C=RGB(255,80+(U*5) MOD 170,80+(V*8) MOD 170,160)
   FA%[I*2]=I:FB%[I*2]=K:FD%[I*2]=E:FK%[I*2]=C
   FA%[I*2+1]=I:FB%[I*2+1]=E:FD%[I*2+1]=J:FK%[I*2+1]=C
  NEXT
 NEXT
 'SPRITES ON TOP, AND A CLOUD BEHIND ON A LAYER THAT GETS BLURRED IN A BURST
 FOR I=0 TO NSP-1
  SPSET I,(I AND 7)*16,16,16,16,0
  SPOFS I,RX%[I],RY%[I],-100
 NEXT
 LAYER 2,1
 FOR I=0 TO 19
  SPSET 3000+I,0,64,64,64,0
  SPLAYER 3000+I,2
  SPOFS 3000+I,RX%[I],RY%[I]
 NEXT
 TSCREEN 0,8,8,50,30
 GCLIP 0,0,399,239
 GO NVT
 WHILE TF()
  P=FC AND 1
  GTARGET P
  GCLS
  CA#=COS(FC*0.02):SA#=SIN(FC*0.02)
  CB#=COS(FC*0.013):SB#=SIN(FC*0.013)
  FOR I=0 TO NVT-1
   X0#=VX#[I]:Y0#=VY#[I]:Z0#=VZ#[I]
   X1#=X0#*CA#+Z0#*SA#
   Z1#=Z0#*CA#-X0#*SA#
   Y1#=Y0#*CB#-Z1#*SB#
   Z1#=Y0#*SB#+Z1#*CB#+300
   W#=220/Z1#
   SX#[I]=200+X1#*W#
   SY#[I]=120+Y1#*W#
   SZ#[I]=Z1#
  NEXT
  C=0
  FOR I=0 TO NFA-1
   A=FA%[I]:B=FB%[I]:D=FD%[I]
   X0#=SX#[A]:Y0#=SY#[A]
   IF (SX#[B]-X0#)*(SY#[D]-Y0#)-(SY#[B]-Y0#)*(SX#[D]-X0#)>0 THEN
    DEPTH%[C]=-SZ#[A]
    ORD%[C]=I
    INC C
   ENDIF
  NEXT
  SORT 0,C,DEPTH%,ORD%
  FOR K=0 TO C-1
   I=ORD%[K]
   A=FA%[I]:B=FB%[I]:D=FD%[I]
   GTRI SX#[A],SY#[A],SX#[B],SY#[B],SX#[D],SY#[D],FK%[I]
  NEXT
  FOR I=0 TO NSP-1
   SPOFS I,RX%[(I+FC) AND 8191],RY%[I]
  NEXT
  LOCATE 0,0,27
  TPRINT 0,"FRAME ";FC;" FACES ";C;"   ";
  'BURST: EXTRA TRIANGLES
  IF (FC MOD 60)==59 THEN
   FOR I=0 TO BT-1
    J=(I+FC) AND 8191
    X=RX%[J]:Y=RY%[J]
    GTRI X,Y,X+24,Y+4,X+8,Y+22,RC%[J]
   NEXT
  ENDIF
  'BURST: SPRITES APPEAR, AND GO LATER
  IF (FC MOD 120)==30 THEN
   FOR I=0 TO BS-1
    J=(I+FC) AND 8191
    SPSET 1000+I,(I AND 7)*16,16,16,16,0
    SPOFS 1000+I,RX%[J],RY%[J],-200
    SPCOLOR 1000+I,RC%[J]
   NEXT
  ENDIF
  IF (FC MOD 120)==50 THEN SPCLR 1000,999+BS
  'BURST: THE BLUR COMES ON, AND THE TEXT IS REWRITTEN
  IF (FC MOD 180)==90 THEN
   LFILTER 2,2,32
   BLUR=1
   TFILL 0,0,0,49,26,33+(FC AND 63)
  ENDIF
  IF BLUR && (FC MOD 180)==102 THEN
   LFILTER 2,0
   BLUR=0
   TFILL 0,0,0,49,26,32
  ENDIF
  SPPAGE #GSPRITE,P
 WEND
 FIN
END
'
'--- MAX LOAD ---
'THE COUNT GROWS BY A THIRD A STEP UNTIL A STEP FALLS BEHIND 60 FPS, AND IS
'THEN HALVED TOWARDS THE LAST ONE THAT HELD. THE SCORE IS THE LAST COUNT THAT
'HELD. IT DEPENDS ON THE CLOCK, SO WITH FROZEN TIME THE SCENE IS LEFT OUT.
DEF RAMPINIT START,CAP
 MLO=0:MHI=0:MCUR=START:MCAP=CAP:MT0=MILLISEC()
 FLIM=#RAMPFRAMES
 MSLIM=#RAMPSTEPMS
END
'
'JUDGES THE STEP JUST MEASURED AND CHOOSES THE NEXT COUNT, -1 WHEN DONE
DEF RAMPNEXT
 VAR I,S#,OK=0,NXT
 IF FC>=FLIM THEN
  FOR I=0 TO FC-1
   S#=S#+FRM%[I]
  NEXT
  OK=(S#/FC<=#RAMPOK) && (MAINCNT()-M0-FC<=1)
 ENDIF
 IF OK THEN
  MLO=MCUR
 ELSE
  MHI=MCUR
 ENDIF
 IF MHI==0 THEN
  IF MCUR>=MCAP THEN
   NXT=-1
  ELSE
   NXT=MIN(MCAP,MCUR*4 DIV 3)
  ENDIF
 ELSEIF (MHI-MLO)*16<=MHI THEN
  NXT=-1
 ELSE
  NXT=(MLO+MHI) DIV 2
 ENDIF
 IF MILLISEC()-MT0>#RAMPMS THEN NXT=-1
 LOCATE 0,2
 ?"N=";MCUR;" HELD ";MLO;"       ";
 MCUR=NXT
END
'
'THE TABLE'S ROW: THE SCORE IN THE COUNT COLUMN, THE LAST STEP'S TIMES
DEF RAMPEND
 VAR N=FC,R=RN,A#,B#,C#
 RLOST[R]=-1
 RWALL[R]=MILLISEC()-MT0
 RNAME$[R]=NM$[CUR]
 RCNT[R]=MLO
 RFR[R]=N
 IF N>0 THEN
  STATS WORK%,N OUT A#,B#,C#
  RWM#[R]=A#:RWP#[R]=B#:RWX#[R]=C#
  STATS FRM%,N OUT A#,B#,C#
  RFM#[R]=A#:RFP#[R]=B#:RFX#[R]=C#
 ENDIF
 IF FROZEN THEN RCNT[R]=-1
 INC RN
END
'
DEF SCMAXSP
 VAR I,N,HAVE,X,Y,J
 PREP
 GO 0
 FC=0
 IF !FROZEN THEN
  RAMPINIT CN(#MAXSP0),#MAXSPCAP
  WHILE MCUR>=0
   N=MCUR
   FOR I=HAVE TO N-1
    SPSET I,(I AND 3)*32,32,32,32,0
    SPHOME I,16,16
   NEXT
   IF N<HAVE THEN SPCLR N,HAVE-1
   HAVE=N
   STARTCLOCK
   WHILE TF()
    FOR I=0 TO N-1
     J=(I*3+FC) AND 8191
     X=RX%[J]+SINT#[(I+FC) AND 255]*16
     Y=RY%[J]+SINT#[(I+FC+64) AND 255]*16
     SPOFS I,X,Y
     SPROT I,(I+FC*2) MOD 360
     SPSCALE I,0.75+SINT#[(I+FC*2) AND 255]*0.25,0.75
     SPCOLOR I,RC%[I AND 8191]
    NEXT
   WEND
   RAMPNEXT
  WEND
 ENDIF
 RAMPEND
END
'
DEF SCMAXTRI
 VAR I,N,J,X,Y,K
 PREP
 GCLIP 0,0,399,239
 GO 0
 FC=0
 IF !FROZEN THEN
  RAMPINIT CN(#MAXTRI0),#MAXTRICAP
  WHILE MCUR>=0
   N=MCUR
   STARTCLOCK
   WHILE TF()
    GCLS
    K=FC*37
    FOR I=0 TO N-1
     J=(I+K) AND 8191
     X=RX%[J]:Y=RY%[J]
     GTRI X,Y,X+24,Y+4,X+8,Y+22,RC%[J]
    NEXT
   WEND
   RAMPNEXT
  WEND
 ENDIF
 RAMPEND
END
'
'--- THE RESULTS ---
DEF PAD(S$,N)
 IF LEN(S$)>=N THEN RETURN LEFT$(S$,N)
 RETURN S$+" "*(N-LEN(S$))
END
'
'A TIME IN SIX COLUMNS
DEF T6(V#)
 IF V#>=1000 THEN RETURN FORMAT$("%6.0F",V#)
 RETURN FORMAT$("%6.2F",V#)
END
'
DEF RESULTTEXT()
 VAR S$,I,NL$=CHR$(10)
 S$="TPERF 1"+NL$
 S$=S$+"TAG "+#TAG+NL$
 S$=S$+"LOADPCT "+STR$(#LOADPCT)+NL$
 S$=S$+"FRAMES "+STR$(#FRAMES)+NL$
 S$=S$+"ONLY "+STR$(#ONLY)+NL$
 S$=S$+"TVMODE "+STR$(SYSPARAM("TVMODE"))+NL$
 S$=S$+"CLOCK "+STR$(!FROZEN)+NL$
 S$=S$+"DATE "+DATE$()+" "+TIME$()+NL$
 S$=S$+"TOTALMS "+STR$(TOTALMS)+NL$
 S$=S$+"SCENE COUNT FRAMES WMEAN WP95 WMAX FMEAN FP95 FMAX LOST WALLMS"+NL$
 FOR I=0 TO RN-1
  S$=S$+RNAME$[I]+" "+STR$(RCNT[I])+" "+STR$(RFR[I])
  S$=S$+FORMAT$(" %.3F %.3F %.3F",RWM#[I],RWP#[I],RWX#[I])
  S$=S$+FORMAT$(" %.3F %.3F %.3F",RFM#[I],RFP#[I],RFX#[I])
  S$=S$+" "+STR$(RLOST[I])+" "+STR$(RWALL[I])+NL$
 NEXT
 RETURN S$
END
'
DEF SAVERESULT
 VAR S$=RESULTTEXT()
 SAVED$="SAVING: ANSWER YES TO SAVE THE RESULTS"
 SHOWPAGE
 VSYNC
 SAVEV "TXT:"+#RESFILE,S$
 IF RESULT()==1 THEN
  SAVED$="SAVED AS TXT:"+#RESFILE
 ELSE
  SAVED$="NOT SAVED: PRESS Y, OR TAKE A SCREENSHOT"
 ENDIF
 SHOWPAGE
END
'
'ONE SCREEN OF THE TABLE. WORK AND FRAME ARE IN MS; LOST IS FRAMES
DEF SHOWPAGE
 VAR I,R,ROWS=23
 CLS
 LOCATE 0,0
 ?"TPERF  PAGE ";PAGE+1;"/";PAGES;"  TOTAL ";TOTALMS DIV 1000;" S"
 ?SAVED$
 ?"         --------WORK-------- ---FRAME---"
 ?"SCENE      MEAN    P95  WORST   MEAN  WORST LOST"
 FOR R=0 TO ROWS-1
  I=PAGE*ROWS+R
  IF I<RN THEN
   IF RLOST[I]<0 THEN
    ?PAD(RNAME$[I],9);
    IF RCNT[I]<0 THEN
     ?"NOT RUN: TIME IS FROZEN"
    ELSE
     ?"SCORE ";RCNT[I];" AT 60 FPS"
    ENDIF
   ELSE
    ?PAD(RNAME$[I],8);" ";T6(RWM#[I]);" ";T6(RWP#[I]);" ";T6(RWX#[I]);" ";
    ?T6(RFM#[I]);" ";T6(RFX#[I]);FORMAT$("%5D",INT(RLOST[I]))
   ENDIF
  ENDIF
 NEXT
 LOCATE 0,28
 ?"TIMES IN MS. A:NEXT PAGE  Y:SAVE AGAIN  B:END";
END
'
DEF FINISHRUN
 VAR K$
 ACLS 1,1,1,1
 GCLS
 PAGES=(MAX(0,RN-1) DIV 23)+1
 PAGE=0
 SAVERESULT
 WHILE 1
  VSYNC
  K$=INKEY$()
  IF BUTTON(0,#B_RDOWN,1) THEN BREAK
  IF BUTTON(0,#B_RLEFT,1) THEN SAVERESULT
  IF BUTTON(0,#B_RRIGHT,1) || K$!="" THEN
   PAGE=(PAGE+1) MOD PAGES
   SHOWPAGE
  ENDIF
 WEND
END
